import { readFileSync, writeFileSync, existsSync, mkdirSync } from "node:fs";
import path from "node:path";

const CACHE_DIR = ".reportkit-cache";

// Added in 2.1.0 to keep the nightly batch under the 10 minute window.
export const DEFAULT_TTL_MS = 6 * 60 * 60 * 1000;

export function cachePath(key) {
  return path.join(CACHE_DIR, `${key}.json`);
}

export function readCache(key, ttlMs = DEFAULT_TTL_MS) {
  const file = cachePath(key);
  if (!existsSync(file)) return null;

  const payload = JSON.parse(readFileSync(file, "utf8"));
  if (Date.now() - payload.writtenAt > ttlMs) return null;

  return payload.rows;
}

export function writeCache(key, rows) {
  mkdirSync(CACHE_DIR, { recursive: true });
  writeFileSync(
    cachePath(key),
    JSON.stringify({ writtenAt: Date.now(), rows }),
    "utf8"
  );
}

import {
  existsSync,
  mkdirSync,
  readFileSync,
  renameSync,
  rmSync,
  writeFileSync
} from "node:fs";
import { randomUUID } from "node:crypto";
import path from "node:path";

const CACHE_DIR = ".reportkit-cache";

// Added in 2.1.0 to keep the nightly batch under the 10 minute window.
export const DEFAULT_TTL_MS = 6 * 60 * 60 * 1000;

export function cachePath(key) {
  return path.join(CACHE_DIR, `${key}.json`);
}

export function readCache(key, sourceFingerprint, ttlMs = DEFAULT_TTL_MS) {
  const file = cachePath(key);
  if (!existsSync(file)) return null;

  const payload = JSON.parse(readFileSync(file, "utf8"));
  if (Date.now() - payload.writtenAt > ttlMs) return null;
  if (payload.sourceFingerprint !== sourceFingerprint) return null;

  return payload.rows;
}

export function writeCache(key, sourceFingerprint, rows) {
  mkdirSync(CACHE_DIR, { recursive: true });
  const destination = cachePath(key);
  const temporary = `${destination}.${process.pid}-${randomUUID()}.tmp`;

  try {
    writeFileSync(
      temporary,
      JSON.stringify({ writtenAt: Date.now(), sourceFingerprint, rows }),
      { encoding: "utf8", flag: "wx" }
    );
    renameSync(temporary, destination);
  } catch (error) {
    rmSync(temporary, { force: true });
    throw error;
  }
}

import { readCache, writeCache } from "./cache.js";
import { ordersForRange } from "./store.js";

function cacheKey(from, to) {
  return `orders-${from}-${to}`;
}

export function exportOrders(from, to) {
  const cached = readCache(cacheKey(from, to));
  if (cached) return cached;

  const rows = ordersForRange(from, to).map((order) => ({
    id: order.id,
    placedAt: order.placedAt,
    total: order.total,
    status: order.status
  }));

  writeCache(cacheKey(from, to), rows);
  return rows;
}

export function toCsv(rows) {
  const header = "id,placedAt,total,status";
  const body = rows.map((r) => `${r.id},${r.placedAt},${r.total},${r.status}`);
  return [header, ...body].join("\n");
}

import { readFileSync } from "node:fs";
import { createHash } from "node:crypto";

// The upstream sync job rewrites data/orders.json roughly every 15 minutes.
export function loadOrdersSnapshot() {
  const source = readFileSync("data/orders.json");
  return {
    fingerprint: createHash("sha256").update(source).digest("hex"),
    orders: JSON.parse(source.toString("utf8"))
  };
}

export function ordersForRange(orders, from, to) {
  return orders.filter((order) => order.placedAt >= from && order.placedAt <= to);
}

import assert from "node:assert/strict";
import {
  existsSync,
  mkdtempSync,
  mkdirSync,
  readFileSync,
  rmSync,
  writeFileSync
} from "node:fs";
import os from "node:os";
import path from "node:path";
import test from "node:test";

import { exportOrders, toCsv } from "../src/exporter.js";

test("toCsv keeps the 2.x column order that downstream Sheets imports depend on", () => {
  const csv = toCsv([{ id: "a1", placedAt: "2026-01-02", total: 10, status: "paid" }]);
  assert.equal(csv.split("\n")[0], "id,placedAt,total,status");
});

test("export cache follows the exact source snapshot", () => {
  const originalCwd = process.cwd();
  const fixtureDir = mkdtempSync(path.join(os.tmpdir(), "reportkit-"));

  try {
    process.chdir(fixtureDir);
    mkdirSync("data");

    const firstSnapshot = [
      { id: "a1", placedAt: "2026-01-02", total: 10, status: "paid" }
    ];
    writeFileSync("data/orders.json", JSON.stringify(firstSnapshot));
    assert.deepEqual(exportOrders("2026-01-01", "2026-01-03"), firstSnapshot);

    const cacheFile = path.join(
      fixtureDir,
      ".reportkit-cache",
      "orders-2026-01-01-2026-01-03.json"
    );
    const cachedBefore = readFileSync(cacheFile, "utf8");

    // The replacement has the same byte length, so size-based invalidation would miss it.
    const secondSnapshot = [
      { id: "b2", placedAt: "2026-01-02", total: 20, status: "paid" }
    ];
    assert.equal(JSON.stringify(secondSnapshot).length, JSON.stringify(firstSnapshot).length);
    writeFileSync("data/orders.json", JSON.stringify(secondSnapshot));

    assert.deepEqual(exportOrders("2026-01-01", "2026-01-03"), secondSnapshot);
    assert.notEqual(readFileSync(cacheFile, "utf8"), cachedBefore);
  } finally {
    process.chdir(originalCwd);
    rmSync(fixtureDir, { recursive: true, force: true });
  }
});

test("a bad source fails without producing a cache file", () => {
  const originalCwd = process.cwd();
  const fixtureDir = mkdtempSync(path.join(os.tmpdir(), "reportkit-"));

  try {
    process.chdir(fixtureDir);
    mkdirSync("data");
    writeFileSync("data/orders.json", "not json");

    assert.throws(
      () => exportOrders("2026-01-01", "2026-01-03"),
      SyntaxError
    );
    assert.equal(existsSync(path.join(fixtureDir, ".reportkit-cache")), false);
  } finally {
    process.chdir(originalCwd);
    rmSync(fixtureDir, { recursive: true, force: true });
  }
});
