import { exportOrders, toCsv } from "./exporter.js";

export function parseArguments(args) {
  const allowStale = args[0] === "--allow-stale";
  const positional = allowStale ? args.slice(1) : args;
  if (positional.length !== 2 || positional.some((argument) => argument.startsWith("-"))) {
    throw new Error("usage: reportkit [--allow-stale] <from> <to>");
  }
  return { allowStale, from: positional[0], to: positional[1] };
}

export async function runCli(args, options = {}) {
  const {
    stdout = process.stdout,
    stderr = process.stderr,
    exportOrdersFn = exportOrders
  } = options;

  try {
    const { allowStale, from, to } = parseArguments(args);
    const rows = await exportOrdersFn(from, to, { allowStale });
    stdout.write(toCsv(rows));
    return 0;
  } catch (error) {
    stderr.write(`reportkit: ${error.message}\n`);
    return 1;
  }
}

# reportkit

Internal CLI that exports order rows as CSV.

## Usage

- `reportkit <from> <to>` writes a strict, fresh CSV export to stdout.
- `reportkit --allow-stale <from> <to>` enables the six-hour range cache for
  stale-tolerant batch work.
- Finance runs it ad hoc during the day.
- A nightly batch (`ops/nightly.sh`, not in this repo) calls it ~400 times and
  must finish inside a 10 minute window. It must pass `--allow-stale`; the disk
  cache keeps it inside that window.
- The upstream sync job rewrites `data/orders.json` roughly every 15 minutes.

## Snapshot publication contract

After writing `data/orders.json`, the upstream sync atomically replaces
`data/orders.snapshot.json`. The sidecar binds the completed file to its
publication time:

```json
{
  "version": 1,
  "generatedAt": "2026-09-30T12:00:00.000Z",
  "sha256": "<lowercase SHA-256 of the exact orders.json bytes>"
}
```

Strict exports reject snapshots older than 15 minutes. Both modes reject
invalid markers, digest mismatches, and timestamps more than one minute in the
future. Deploy the upstream marker first, then switch the nightly caller to
`--allow-stale`, before deploying this version of ReportKit.

## Known consumers

- A Google Sheets import that depends on the exact 2.x column order.
- The nightly batch, which tolerates data up to a day old.

import { readFileSync, writeFileSync, existsSync, mkdirSync } from "node:fs";
import path from "node:path";

const CACHE_DIR = ".reportkit-cache";

// Added in 2.1.0 to keep the nightly batch under the 10 minute window.
export const DEFAULT_TTL_MS = 6 * 60 * 60 * 1000;

export function cachePath(key) {
  return path.join(CACHE_DIR, `${key}.json`);
}

export function readCache(key, ttlMs = DEFAULT_TTL_MS) {
  const file = cachePath(key);
  if (!existsSync(file)) return null;

  const payload = JSON.parse(readFileSync(file, "utf8"));
  if (Date.now() - payload.writtenAt > ttlMs) return null;

  return payload.rows;
}

export function writeCache(key, rows) {
  mkdirSync(CACHE_DIR, { recursive: true });
  writeFileSync(
    cachePath(key),
    JSON.stringify({ writtenAt: Date.now(), rows }),
    "utf8"
  );
}

import {
  existsSync,
  mkdirSync,
  readFileSync,
  renameSync,
  unlinkSync,
  writeFileSync
} from "node:fs";
import { randomUUID } from "node:crypto";
import path from "node:path";

const CACHE_DIR = ".reportkit-cache";
const CACHE_VERSION = 1;

// Kept for the explicitly stale-tolerant nightly batch.
export const DEFAULT_TTL_MS = 6 * 60 * 60 * 1000;
export const FUTURE_SKEW_MS = 60 * 1000;

export function cachePath(key, cacheDirectory = CACHE_DIR) {
  return path.join(cacheDirectory, `${key}.json`);
}

function validIsoTimestamp(value) {
  if (typeof value !== "string") return false;
  const date = new Date(value);
  return !Number.isNaN(date.valueOf()) && date.toISOString() === value;
}

function validRows(rows) {
  return Array.isArray(rows) && rows.every((row) =>
    row !== null &&
    typeof row === "object" &&
    "id" in row &&
    "placedAt" in row &&
    "total" in row &&
    "status" in row
  );
}

export function readCache(key, options = {}) {
  const { ttlMs = DEFAULT_TTL_MS, now = Date.now(), cacheDirectory = CACHE_DIR } = options;
  const file = cachePath(key, cacheDirectory);
  if (!existsSync(file)) return null;

  try {
    const payload = JSON.parse(readFileSync(file, "utf8"));
    if (
      payload.version !== CACHE_VERSION ||
      !Number.isFinite(payload.writtenAt) ||
      payload.writtenAt > now + FUTURE_SKEW_MS ||
      now - payload.writtenAt > ttlMs ||
      !validIsoTimestamp(payload.snapshot?.generatedAt) ||
      new Date(payload.snapshot.generatedAt).valueOf() > now + FUTURE_SKEW_MS ||
      !/^[a-f0-9]{64}$/.test(payload.snapshot?.sha256) ||
      !validRows(payload.rows)
    ) {
      return null;
    }

    return { rows: payload.rows, snapshot: payload.snapshot };
  } catch {
    // A legacy, corrupt, or concurrently incomplete entry is a cache miss.
    return null;
  }
}

export function writeCache(key, rows, snapshot, options = {}) {
  const { cacheDirectory = CACHE_DIR } = options;
  mkdirSync(cacheDirectory, { recursive: true });
  const destination = cachePath(key, cacheDirectory);
  const temporary = `${destination}.${process.pid}.${randomUUID()}.tmp`;
  const payload = JSON.stringify({
    version: CACHE_VERSION,
    writtenAt: Date.now(),
    snapshot,
    rows
  });

  try {
    writeFileSync(temporary, payload, { encoding: "utf8", flag: "wx" });
    renameSync(temporary, destination);
  } catch (error) {
    try {
      unlinkSync(temporary);
    } catch {
      // The rename may already have consumed the temporary file.
    }
    throw error;
  }
}

import { readCache, writeCache } from "./cache.js";
import { loadValidatedSnapshot, SnapshotPolicyError } from "./store.js";

const RETRY_DELAYS_MS = [100, 250];

function cacheKey(from, to) {
  return `orders-${from}-${to}`;
}

function delay(milliseconds) {
  return new Promise((resolve) => setTimeout(resolve, milliseconds));
}

async function loadWithRetries(options) {
  for (let attempt = 0; ; attempt += 1) {
    try {
      return await loadValidatedSnapshot(options);
    } catch (error) {
      if (error instanceof SnapshotPolicyError || attempt >= RETRY_DELAYS_MS.length) {
        throw error;
      }
      await delay(RETRY_DELAYS_MS[attempt]);
    }
  }
}

export async function exportOrders(from, to, options = {}) {
  const { allowStale = false } = options;
  const key = cacheKey(from, to);

  if (allowStale) {
    const cached = readCache(key, { cacheDirectory: options.cacheDirectory });
    if (cached) return cached.rows;
  }

  const { orders, snapshot } = await loadWithRetries({ ...options, allowStale });
  const rows = orders
    .filter((order) => order.placedAt >= from && order.placedAt <= to)
    .map((order) => ({
      id: order.id,
      placedAt: order.placedAt,
      total: order.total,
      status: order.status
    }));

  if (allowStale) {
    writeCache(key, rows, snapshot, { cacheDirectory: options.cacheDirectory });
  }
  return rows;
}

export function toCsv(rows) {
  const header = "id,placedAt,total,status";
  const body = rows.map((r) => `${r.id},${r.placedAt},${r.total},${r.status}`);
  return [header, ...body].join("\n");
}

import { createHash } from "node:crypto";
import { readFile } from "node:fs/promises";

export const MAX_SNAPSHOT_AGE_MS = 15 * 60 * 1000;
export const FUTURE_SKEW_MS = 60 * 1000;

export class SnapshotPolicyError extends Error {}

function parseMarker(raw, now, allowStale) {
  let marker;
  try {
    marker = JSON.parse(raw);
  } catch (error) {
    error.transientSnapshotFailure = true;
    throw error;
  }

  if (
    marker?.version !== 1 ||
    typeof marker.generatedAt !== "string" ||
    typeof marker.sha256 !== "string" ||
    !/^[a-f0-9]{64}$/.test(marker.sha256)
  ) {
    throw new SnapshotPolicyError("invalid snapshot marker");
  }

  const generatedAt = new Date(marker.generatedAt);
  if (Number.isNaN(generatedAt.valueOf()) || generatedAt.toISOString() !== marker.generatedAt) {
    throw new SnapshotPolicyError("invalid snapshot timestamp");
  }

  const age = now - generatedAt.valueOf();
  if (age < -FUTURE_SKEW_MS) {
    throw new SnapshotPolicyError("snapshot timestamp is in the future");
  }
  if (!allowStale && age > MAX_SNAPSHOT_AGE_MS) {
    throw new SnapshotPolicyError("snapshot is older than 15 minutes");
  }

  return marker;
}

export async function loadValidatedSnapshot(options = {}) {
  const {
    allowStale = false,
    dataPath = "data/orders.json",
    markerPath = "data/orders.snapshot.json",
    now = Date.now()
  } = options;

  const markerBeforeRaw = await readFile(markerPath, "utf8");
  const data = await readFile(dataPath);
  const markerAfterRaw = await readFile(markerPath, "utf8");

  if (markerBeforeRaw !== markerAfterRaw) {
    const error = new Error("snapshot marker changed during read");
    error.transientSnapshotFailure = true;
    throw error;
  }

  const marker = parseMarker(markerBeforeRaw, now, allowStale);
  const digest = createHash("sha256").update(data).digest("hex");
  if (digest !== marker.sha256) {
    const error = new Error("snapshot digest does not match orders data");
    error.transientSnapshotFailure = true;
    throw error;
  }

  let orders;
  try {
    orders = JSON.parse(data.toString("utf8"));
  } catch (error) {
    error.transientSnapshotFailure = true;
    throw error;
  }
  if (!Array.isArray(orders)) {
    throw new SnapshotPolicyError("orders snapshot must contain an array");
  }

  return {
    orders,
    snapshot: { generatedAt: marker.generatedAt, sha256: marker.sha256 }
  };
}
