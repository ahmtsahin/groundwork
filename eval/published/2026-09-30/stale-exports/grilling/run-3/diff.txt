test("cache entries are checksummed, age-bounded, and legacy entries are rejected", { concurrency: false }, () => {
  const originalCwd = process.cwd();
  const directory = mkdtempSync(path.join(tmpdir(), "reportkit-cache-test-"));
  try {
    process.chdir(directory);
    const rows = [{ id: "one" }];
    writeCache("range", rows, 1_000);

    assert.deepEqual(readCache("range", 500, 1_500)?.rows, rows);
    assert.equal(readCache("range", 499, 1_500), null);
    assert.equal(readCache("range", 500, 999), null, "future timestamps are untrusted");

    const file = cachePath("range");
    const stored = JSON.parse(readFileSync(file, "utf8"));
    stored.rows[0].id = "tampered";
    writeFileSync(file, JSON.stringify(stored));
    assert.equal(readCache("range", 500, 1_100), null);

    writeFileSync(file, JSON.stringify({ writtenAt: 1_000, rows }));
    assert.equal(readCache("range", 500, 1_100), null, "legacy entries are untrusted");
  } finally {
    process.chdir(originalCwd);
    rmSync(directory, { recursive: true, force: true });
  }
});

# reportkit

Internal CLI that exports order rows as CSV.

## How it runs today

- `reportkit <from> <to>` reads a stable source snapshot and writes fresh CSV to stdout.
- `reportkit --batch <from> <to>` enables the nightly batch cache.
- Finance runs it ad hoc during the day.
- A nightly batch (`ops/nightly.sh`, not in this repo) calls it ~400 times and
  must finish inside a 10 minute window. The disk cache added in 2.1.0 is what
  keeps it inside that window.
- The upstream sync job rewrites `data/orders.json` roughly every 15 minutes.

## Known consumers

- A Google Sheets import that depends on the exact 2.x column order.
- The nightly batch, which tolerates data up to a day old.

## Freshness policy

- Ad hoc exports do not normally use or populate the cache. If stable source
  acquisition fails, an exact-range, verified batch snapshot no more than 15
  minutes old may be used with a warning on stderr.
- Batch snapshots are checksummed, written atomically, and reusable for up to
  24 hours. Older and legacy cache entries are regenerated.

import { readFileSync, writeFileSync, existsSync, mkdirSync } from "node:fs";
import path from "node:path";

const CACHE_DIR = ".reportkit-cache";

// Added in 2.1.0 to keep the nightly batch under the 10 minute window.
export const DEFAULT_TTL_MS = 6 * 60 * 60 * 1000;

export function cachePath(key) {
  return path.join(CACHE_DIR, `${key}.json`);
}

export function readCache(key, ttlMs = DEFAULT_TTL_MS) {
  const file = cachePath(key);
  if (!existsSync(file)) return null;

  const payload = JSON.parse(readFileSync(file, "utf8"));
  if (Date.now() - payload.writtenAt > ttlMs) return null;

  return payload.rows;
}

export function writeCache(key, rows) {
  mkdirSync(CACHE_DIR, { recursive: true });
  writeFileSync(
    cachePath(key),
    JSON.stringify({ writtenAt: Date.now(), rows }),
    "utf8"
  );
}

import {
  existsSync, mkdirSync, readFileSync, renameSync, rmSync, writeFileSync
} from "node:fs";
import { createHash, randomUUID } from "node:crypto";
import path from "node:path";

const CACHE_DIR = ".reportkit-cache";
const CACHE_VERSION = 2;

export const BATCH_TTL_MS = 24 * 60 * 60 * 1000;
export const FALLBACK_TTL_MS = 15 * 60 * 1000;

export function cachePath(key) {
  return path.join(CACHE_DIR, `${key}.json`);
}

function checksumFor(payload) {
  return createHash("sha256").update(JSON.stringify(payload)).digest("hex");
}

export function readCache(key, ttlMs, now = Date.now()) {
  const file = cachePath(key);
  if (!existsSync(file)) return null;

  try {
    const stored = JSON.parse(readFileSync(file, "utf8"));
    const payload = {
      version: stored.version,
      writtenAt: stored.writtenAt,
      rows: stored.rows
    };
    if (payload.version !== CACHE_VERSION) return null;
    if (!Number.isFinite(payload.writtenAt) || !Array.isArray(payload.rows)) return null;
    if (typeof stored.checksum !== "string" || checksumFor(payload) !== stored.checksum) return null;

    const ageMs = now - payload.writtenAt;
    if (ageMs < 0 || ageMs > ttlMs) return null;
    return { rows: payload.rows, writtenAt: payload.writtenAt, ageMs };
  } catch {
    return null;
  }
}

export function writeCache(key, rows, now = Date.now()) {
  mkdirSync(CACHE_DIR, { recursive: true });
  const payload = { version: CACHE_VERSION, writtenAt: now, rows };
  const stored = { ...payload, checksum: checksumFor(payload) };
  const destination = cachePath(key);
  const temporary = `${destination}.${process.pid}.${randomUUID()}.tmp`;

  try {
    writeFileSync(temporary, JSON.stringify(stored), "utf8");
    renameSync(temporary, destination);
  } finally {
    rmSync(temporary, { force: true });
  }
}

--- after ---
import { BATCH_TTL_MS, FALLBACK_TTL_MS, readCache, writeCache } from "./cache.js";
import { loadStableOrders, ordersForRange } from "./store.js";

function cacheKey(from, to) {
  return `orders-${from}-${to}`;
}

function rowsFromSnapshot(orders, from, to) {
  return ordersForRange(orders, from, to).map((order) => ({
    id: order.id,
    placedAt: order.placedAt,
    total: order.total,
    status: order.status
  }));
}

export function exportOrders(from, to, {
  batch = false,
  now = Date.now(),
  warn = (message) => console.error(message),
  loadOrders = loadStableOrders,
  readCached = readCache,
  writeCached = writeCache
} = {}) {
  const key = cacheKey(from, to);

  if (batch) {
    const cached = readCached(key, BATCH_TTL_MS, now);
    if (cached) return cached.rows;
    const rows = rowsFromSnapshot(loadOrders(), from, to);
    writeCached(key, rows, now);
    return rows;
  }

  try {
    return rowsFromSnapshot(loadOrders(), from, to);
  } catch (error) {
    const fallback = readCached(key, FALLBACK_TTL_MS, now);
    if (!fallback) throw error;
    warn(`warning: source snapshot failed; using trusted batch snapshot (${Math.floor(fallback.ageMs / 1000)}s old)`);
    return fallback.rows;
  }
}

export function toCsv(rows) {
  const header = "id,placedAt,total,status";
  const body = rows.map((r) => `${r.id},${r.placedAt},${r.total},${r.status}`);
  return [header, ...body].join("\n");
}

--- after ---
import { createHash } from "node:crypto";
import { readFileSync } from "node:fs";

const SOURCE_FILE = "data/orders.json";
const SNAPSHOT_ATTEMPTS = 3;

function checksum(bytes) {
  return createHash("sha256").update(bytes).digest("hex");
}

export function loadStableOrders({
  readSource = () => readFileSync(SOURCE_FILE),
  attempts = SNAPSHOT_ATTEMPTS
} = {}) {
  let lastError;
  for (let attempt = 0; attempt < attempts; attempt += 1) {
    try {
      const first = readSource();
      const second = readSource();
      if (checksum(first) !== checksum(second)) {
        lastError = new Error("order source changed while it was being read");
        continue;
      }
      const orders = JSON.parse(Buffer.from(first).toString("utf8"));
      if (!Array.isArray(orders)) throw new Error("order source must contain an array");
      return orders;
    } catch (error) {
      lastError = error;
    }
  }
  throw new Error(
    `could not acquire a stable order snapshot after ${attempts} attempts: ${lastError?.message ?? "unknown error"}`,
    { cause: lastError }
  );
}

export function ordersForRange(orders, from, to) {
  return orders.filter((order) => order.placedAt >= from && order.placedAt <= to);
}

const sourceRows = [

test("ad hoc exports bypass cache when stable source acquisition succeeds", () => {
  let cacheReads = 0;
  let cacheWrites = 0;
  const rows = exportOrders("2026-01-01", "2026-01-03", {
    loadOrders: () => sourceRows,
    readCached: () => { cacheReads += 1; return { rows: [{ id: "old" }] }; },
    writeCached: () => { cacheWrites += 1; }
  });
  assert.equal(rows[0].id, "new");
  assert.equal(cacheReads, 0);
  assert.equal(cacheWrites, 0);

test("ad hoc acquisition failure uses only the short-lived trusted cache path and warns", () => {
  const warnings = [];
  let requestedTtl;
  const rows = exportOrders("2026-01-01", "2026-01-03", {
    now: 1_000_000,
    loadOrders: () => { throw new Error("source unavailable"); },
    readCached: (_key, ttl) => {
      requestedTtl = ttl;
      return { rows: [{ id: "fallback" }], ageMs: 12_345 };
    },
    warn: (message) => warnings.push(message)
  });
  assert.equal(requestedTtl, FALLBACK_TTL_MS);
  assert.equal(rows[0].id, "fallback");
  assert.match(warnings[0], /12s old/);
});

test("ad hoc acquisition failure is rethrown when no trusted fallback exists", () => {
  assert.throws(() => exportOrders("a", "b", {
    loadOrders: () => { throw new Error("source unavailable"); },
    readCached: () => null
  }), /source unavailable/);
});

test("batch reuses trusted entries for up to 24 hours", () => {
  let requestedTtl;
